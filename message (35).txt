local function parseValue(text)
    text = tostring(text or ""):gsub("%s", "")
    local num, suffix = text:match("([%d%.]+)([KkMmBbTt]?)")
    if not num then return 0 end
    num = tonumber(num) or 0
    local multipliers = {K=1e3, M=1e6, B=1e9, T=1e12}
    local mult = multipliers[(suffix or ""):upper()] or 1
    return num * mult
end

local SAVE_FILE    = "game_settings_v1.json"

-- Default config — every setting that should persist goes here
local Config = {
    Tool           = "Flying Carpet",
    TpKey          = "T",
    CloneKey       = "V",
    AutoTPPriority = true,
    TargetingMode  = "priority",
    StealDelay     = 0,

-- Convert stored key name strings back to KeyCode enums after load
local function resolveKeys()
    Config.TpKey    = Enum.KeyCode[Config.TpKey]    or Enum.KeyCode.T
    Config.CloneKey = Enum.KeyCode[Config.CloneKey] or Enum.KeyCode.V
end

local function saveConfig()
    if not _writefile then return end
    pcall(function()
        local data = {
            Tool           = Config.Tool,
            TpKey          = Config.TpKey.Name,
            CloneKey       = Config.CloneKey.Name,
            AutoTPPriority = Config.AutoTPPriority,
            TargetingMode  = Config.TargetingMode,
            StealDelay     = Config.StealDelay,
            AntiDieOn      = Config.AntiDieOn,
            PriorityList   = PRIORITY_LIST,
        }
        _writefile(SAVE_FILE, game:GetService("HttpService"):JSONEncode(data))
    end)
end

local function loadConfig()
    if not _readfile then return end
    if _isfile and not _isfile(SAVE_FILE) then return end
    pcall(function()
        local raw = _readfile(SAVE_FILE)
        if not raw or raw == "" then return end
        local ok, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(raw)
        end)
        if not ok or type(data) ~= "table" then return end

        if type(data.Tool)           == "string"  then Config.Tool           = data.Tool           end
        if type(data.TpKey)          == "string"  then Config.TpKey          = data.TpKey          end
        if type(data.CloneKey)       == "string"  then Config.CloneKey       = data.CloneKey       end
        if type(data.AutoTPPriority) == "boolean" then Config.AutoTPPriority = data.AutoTPPriority end
        if type(data.TargetingMode)  == "string"  then Config.TargetingMode  = data.TargetingMode  end
        if type(data.StealDelay)     == "number"  then Config.StealDelay     = math.clamp(data.StealDelay, 0, 2) end
        if type(data.AntiDieOn)      == "boolean" then Config.AntiDieOn      = data.AntiDieOn      end

        if type(data.PriorityList) == "table" and #data.PriorityList > 0 then
            for i = #PRIORITY_LIST, 1, -1 do PRIORITY_LIST[i] = nil end
            for _, v in ipairs(data.PriorityList) do table.insert(PRIORITY_LIST, v) end
        end
    end)
end

local function walkForward(seconds)
    local char = LocalPlayer.Character
    local hum  = char:FindFirstChild("Humanoid")
    local hrp  = char:FindFirstChild("HumanoidRootPart")
    local ctrl = getControls()
    local look = hrp.CFrame.LookVector
    ctrl:Disable()
    local startTime = os.clock()
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if os.clock() - startTime >= seconds then
            conn:Disconnect()
            hum:Move(Vector3.zero, false)
            ctrl:Enable()
            return
        end
        hum:Move(look, false)
    end)
end

        if not (char and hum) then error("no character") end

            LocalPlayer.Backpack:FindFirstChild("Quantum Cloner")
            or char:FindFirstChild("Quantum Cloner")

        pcall(function() hum:EquipTool(cloner) end)
        task.wait(0.05)
        cloner:Activate()

            if Workspace:FindFirstChild(cloneName) then break end

        local qcFrame     = toolsFrames and toolsFrames:FindFirstChild("QuantumCloner")

            firesignal(tpButton.MouseButton1Up)

local function fAdornee(animalData)
    if not animalData then return nil end
    local plots = Workspace:FindFirstChild("Plots")
    if not plots then return nil end
    local plot = plots:FindFirstChild(animalData.plot)
    if not plot then return nil end
    local podiums = plot:FindFirstChild("AnimalPodiums")
    if not podiums then return nil end
    local podium = podiums:FindFirstChild(animalData.slot)
    if not podium then return nil end
    local base = podium:FindFirstChild("Base")
    if not base then return nil end
    local spawn = base:FindFirstChild("Spawn")
    if spawn then return spawn end
    return base:FindFirstChildWhichIsA("BasePart") or base
end

local function isPlotUnlocked(plotName)
    local ok, res = pcall(function()
        local plots = Workspace:FindFirstChild("Plots")
        if not plots then return false end
        local targetPlot = plots:FindFirstChild(plotName)
        if not targetPlot then return false end
        local unlockFolder = targetPlot:FindFirstChild("Unlock")
        if not unlockFolder then return true end
        local items = {}
        for _, item in pairs(unlockFolder:GetChildren()) do
            local pos
            if item:IsA("Model") then
                pcall(function() pos = item:GetPivot().Position end)
            elseif item:IsA("BasePart") then
                pos = item.Position
            end
            if pos then table.insert(items, {Object=item, Height=pos.Y}) end
        end
        table.sort(items, function(a,b) return a.Height < b.Height end)
        if #items == 0 then return true end
        local door = items[1].Object
        for _, desc in ipairs(door:GetDescendants()) do
            if desc:IsA("ProximityPrompt") and desc.Enabled then return false end
        end
        for _, child in ipairs(door:GetChildren()) do
            if child:IsA("ProximityPrompt") and child.Enabled then return false end
        end
        return true
    end)
    return ok and res or false
end

local function getClosestBaseIdx(pos)
    local closest, dist = 1, math.huge
    for i, bp in pairs(BASES_LOW) do
        local d = (Vector2.new(pos.X, pos.Z) - Vector2.new(bp.X, bp.Z)).Magnitude
        if d < dist then dist = d; closest = i end
    end
    return closest
end

local function fbpd()
    local debris = Workspace:FindFirstChild("Debris")
    if not debris then return nil end
    local bestPet = { value = -1, part = nil }
    for _, template in ipairs(debris:GetChildren()) do
        if template.Name == "FastOverheadTemplate" then
            local surfaceGui = template:FindFirstChildOfClass("SurfaceGui")
            if surfaceGui and surfaceGui.Adornee and surfaceGui.Adornee.Parent then
                local genLabel = surfaceGui:FindFirstChild("Generation", true)
                if genLabel and genLabel:IsA("TextLabel") then
                    local text = genLabel.Text or ""
                    if text ~= "" and (text:find("/s") or text:find("K") or text:find("M") or text:find("B") or text:find("T")) then
                        local val = parseValue(text)
                        if val > bestPet.value then
                            bestPet.value = val
                            bestPet.part = surfaceGui.Adornee
                        end
                    end
                end
            end
        end
    end
    if bestPet.part then
        return { position = bestPet.part.Position, baseName = getClosestBaseIdx(bestPet.part.Position) }
    end
    return nil
end

local function madp(pos, maxDist)
    maxDist = maxDist or 20
    local best, bestD = nil, math.huge
    for _, a in ipairs(AAC) do
        if a and a.owner ~= LocalPlayer.Name then
            local adornee = fAdornee(a)
            if adornee then
                local d = (adornee.Position - pos).Magnitude
                if d < bestD and d <= maxDist then bestD = d; best = a end
            end
        end
    end
    return best
end

-- ─── Steal system ─────────────────────────────────────────────────────────────
local PMC  = {}
local ISC = {}

local function fppfa(animalData)
    if not animalData then return nil end
    local cp = PMC[animalData.uid]
    if cp and cp.Parent then return cp end
    local plots = Workspace:FindFirstChild("Plots")
    if not plots then return nil end
    local plot = plots:FindFirstChild(animalData.plot)
    if not plot then return nil end
    local podiums = plot:FindFirstChild("AnimalPodiums")
    if not podiums then return nil end
    local foundPodium = podiums:FindFirstChild(animalData.slot)
    if foundPodium then
        local base  = foundPodium:FindFirstChild("Base")
        local spawn = base and base:FindFirstChild("Spawn")
        if spawn then
            local attach = spawn:FindFirstChild("PromptAttachment")
            if attach then
                for _, p in ipairs(attach:GetChildren()) do
                    if p:IsA("ProximityPrompt") and p.Enabled and p.ActionText == "Steal" then
                        PMC[animalData.uid] = p
                        return p
                    end
                end
            end
            local startPos = spawn.Position
            local slotX, slotZ = startPos.X, startPos.Z
            local nearestPrompt, minDist = nil, math.huge
            for _, desc in pairs(plot:GetDescendants()) do
                if desc:IsA("ProximityPrompt") and desc.Enabled and desc.ActionText == "Steal" then
                    local part = desc.Parent
                    local promptPos
                    if part and part:IsA("BasePart") then
                        promptPos = part.Position
                    elseif part and part:IsA("Attachment") and part.Parent and part.Parent:IsA("BasePart") then
                        promptPos = part.Parent.Position
                    end
                    if promptPos then
                        local hDist = math.sqrt((promptPos.X-slotX)^2+(promptPos.Z-slotZ)^2)
                        if hDist < 5 and promptPos.Y > startPos.Y then
                            local yDist = promptPos.Y - startPos.Y
                            if yDist < minDist then minDist = yDist; nearestPrompt = desc end
                        end
                    end
                end
            end
            if nearestPrompt then
                PMC[animalData.uid] = nearestPrompt
                return nearestPrompt
            end
        end
    end
    return nil
end

local function bsc(prompt)
    if ISC[prompt] then return end
    local data = { holdCallbacks = {}, triggerCallbacks = {}, holdEndCallbacks = {} }
    local ok1, conns1 = pcall(getconnections, prompt.PromptButtonHoldBegan)
    if ok1 and type(conns1) == "table" then
        for _, conn in ipairs(conns1) do
            if type(conn.Function) == "function" then table.insert(data.holdCallbacks, conn.Function) end
        end
    end
    local ok2, conns2 = pcall(getconnections, prompt.Triggered)
    if ok2 and type(conns2) == "table" then
        for _, conn in ipairs(conns2) do
            if type(conn.Function) == "function" then table.insert(data.triggerCallbacks, conn.Function) end
        end
    end
    local ok3, conns3 = pcall(getconnections, prompt.PromptButtonHoldEnded)
    if ok3 and type(conns3) == "table" then
        for _, conn in ipairs(conns3) do
            if type(conn.Function) == "function" then table.insert(data.holdEndCallbacks, conn.Function) end
        end
    end
    if (#data.holdCallbacks>0) or (#data.triggerCallbacks>0) or (#data.holdEndCallbacks>0) then
        ISC[prompt] = data
    end
end

local listm   = 0
local ISC_CD = 0.04

local function doSteal(prompt)
    if not prompt then return end
    local now = os.clock()
    if now - listm < ISC_CD then return end
    listm = now
    bsc(prompt)
    local data = ISC[prompt]
    if not data then return end
    local oDur = prompt.HoldDuration
    prompt.HoldDuration = 0
    for _, fn in ipairs(data.holdCallbacks) do pcall(fn) end
    for _rep = 1, 6 do
        for _, fn in ipairs(data.triggerCallbacks) do pcall(fn) end
        if fireproximityprompt then fireproximityprompt(prompt) end
    end
    for _, fn in ipairs(data.holdEndCallbacks) do pcall(fn) end
    prompt.HoldDuration = oDur
end

-- ─── doTP ─────────────────────────────────────────────────────────────
local function doTP()
    if isTpMoving then return end

    local targetPetData = nil
    if Config.TargetingMode == "visible" then
        local debrisBest = fbpd()
        if debrisBest then
            targetPetData = madp(debrisBest.position, 25)
            if not targetPetData then
                local plotIdx = debrisBest.baseName
                targetPetData = { plot="Plot_"..tostring(plotIdx), slot="1", uid="debris_"..tostring(plotIdx), name="Debris target" }
            end
        end
    elseif Config.AutoTPPriority then
        for _, pName in ipairs(PRIORITY_LIST) do
            local sn = pName:lower()
            for _, a in ipairs(AAC) do
                if a and a.name and a.name:lower()==sn and a.owner~=LocalPlayer.Name then
                    targetPetData=a; break
                end
            end
            if targetPetData then break end
        end
        if not targetPetData then
            for _, a in ipairs(AAC) do
                if a and a.owner~=LocalPlayer.Name then targetPetData=a; break end
            end
        end
    else
        if SPD then targetPetData=SPD.animalData end
    end
    if not targetPetData then return end

    local function getChar()
        local c   = LocalPlayer.Character
        local h   = c and c:FindFirstChild("HumanoidRootPart")
        local hum = c and c:FindFirstChildOfClass("Humanoid")
        return c, h, hum
    end

    local char, hrp, hum = getChar()
    if not hrp or not hum or hum.Health<=0 then return end
    isTpMoving = true

    local exactPos
    local targetPart = fAdornee(targetPetData)
    if targetPart then
        exactPos = targetPart.Position
    elseif Config.TargetingMode=="visible" and targetPetData.uid and targetPetData.uid:sub(1,6)=="debris_" then
        local debrisBest = fbpd()
        if not debrisBest then isTpMoving=false; return end
        exactPos = debrisBest.position
    else
        isTpMoving=false; return
    end

    local carpetName = Config.Tool
    local function equipCarpet()
        local c, _, hm = getChar()
        if not c or not hm then return end
        local tool = LocalPlayer.Backpack:FindFirstChild(carpetName) or c:FindFirstChild(carpetName)
        if tool then hm:EquipTool(tool) end
    end
    equipCarpet()
    task.wait(0.01)

    local isSecondFloor = exactPos.Y > 10
    local plotIndex     = getClosestBaseIdx(exactPos)

    if isSecondFloor then
        local targetBasePos = BASES_HAHA_BY_IDX[plotIndex] or BASES_HIGH[plotIndex]
        char, hrp, hum = getChar()
        -- Direct CFrame to second floor — no floating from ground
        for i=1,8 do
            if not hrp then break end
            hrp.AssemblyLinearVelocity=Vector3.zero
            hrp.CFrame=CFrame.new(targetBasePos)
            task.wait(0.03); char,hrp,hum=getChar()
            if hrp and (hrp.Position-targetBasePos).Magnitude<4 then break end
        end
        local bestFace,minFD=FACE_TARGETS[1],math.huge
        for _,v in ipairs(FACE_TARGETS) do
            char,hrp,hum=getChar(); if not hrp then break end
            local d=(hrp.Position-v).Magnitude; if d<minFD then minFD=d; bestFace=v end
        end
        task.wait(0.08); char,hrp,hum=getChar()
        if hrp then hrp.CFrame=CFrame.lookAt(hrp.Position,Vector3.new(bestFace.X,hrp.Position.Y,bestFace.Z)) end
        walkForward(0.35); task.wait(0.35); doClone()
        while _G._cs do task.wait() end
    else
        local bestSpot,minD=CLONE_POSITIONS_FLOOR[1],math.huge
        for _,v in ipairs(CLONE_POSITIONS_FLOOR) do
            local d=(exactPos-v).Magnitude; if d<minD then minD=d; bestSpot=v end
        end
        char,hrp,hum=getChar()
        local jumpStart=tick()
        while hrp and hrp.Position.Y<50 and (tick()-jumpStart)<2.5 do
            hrp.AssemblyLinearVelocity=Vector3.new(hrp.AssemblyLinearVelocity.X,300,hrp.AssemblyLinearVelocity.Z)
            RunService.Heartbeat:Wait(); char,hrp,hum=getChar()
        end
        char,hrp,hum=getChar()
        for i=1,5 do
            if not hrp then break end
            hrp.AssemblyLinearVelocity=Vector3.new(hrp.AssemblyLinearVelocity.X,0,hrp.AssemblyLinearVelocity.Z)
            if (hrp.Position-bestSpot).Magnitude>3 then hrp.CFrame=CFrame.new(bestSpot); task.wait(0.03); char,hrp,hum=getChar() end
        end
        local bestFace,minFD=FACE_TARGETS[1],math.huge
        for _,v in ipairs(FACE_TARGETS) do
            char,hrp,hum=getChar(); if not hrp then break end
            local d=(hrp.Position-v).Magnitude; if d<minFD then minFD=d; bestFace=v end
        end
        task.wait(0.08); char,hrp,hum=getChar()
        if hrp then hrp.CFrame=CFrame.lookAt(hrp.Position,Vector3.new(bestFace.X,hrp.Position.Y,bestFace.Z)) end
        if not isPlotUnlocked(targetPetData.plot) then
            walkForward(0.35); task.wait(0.35); doClone()
            while _G._cs do task.wait() end
        end
    end

    task.wait(0.1); equipCarpet()
    char,hrp,hum=getChar()

    local stealPrompt=fppfa(targetPetData)
    if stealPrompt then bsc(stealPrompt) end

    if isSecondFloor then
        -- Float from current 2nd floor position to just below the podium
        local secondFloorY = BASES_HAHA_BY_IDX[plotIndex] and BASES_HAHA_BY_IDX[plotIndex].Y or 17.5
        local approachY = math.max(exactPos.Y - 6, secondFloorY)
        local airPos=Vector3.new(exactPos.X, approachY, exactPos.Z)
        for i=1,8 do
            char,hrp,hum=getChar(); if not hrp then break end
            if LocalPlayer:GetAttribute("Stealing") then break end

-- ─── Animal scanner ───────────────────────────────────────────────────────────
task.spawn(function()
    local Packages     = ReplicatedStorage:WaitForChild("Packages")
    local Datas        = ReplicatedStorage:WaitForChild("Datas")
    local Shared       = ReplicatedStorage:WaitForChild("Shared")
    local Utils        = ReplicatedStorage:WaitForChild("Utils")
    local Synchronizer  = require(Packages:WaitForChild("Synchronizer"))
    local AnimalsData   = require(Datas:WaitForChild("Animals"))
    local AnimalsShared = require(Shared:WaitForChild("Animals"))
    local NumberUtils   = require(Utils:WaitForChild("NumberUtils"))
    local lastHash = {}

    local function scanPlot(plot)
        local ok,ch=pcall(function() return Synchronizer:Get(plot.Name) end)
        if not ok or not ch then return end
        local al=ch:Get("AnimalList")
        local hash=""
        if al then
            for s,d in pairs(al) do
                if type(d)=="table" then hash=hash..tostring(s)..tostring(d.Index)..tostring(d.Mutation or "") end
            end
        end
        if lastHash[plot.Name]==hash then return end
        lastHash[plot.Name]=hash
        for i=#AAC,1,-1 do
            if AAC[i].plot==plot.Name then table.remove(AAC,i) end
        end
        if not al then return end
        local owner=ch:Get("Owner"); if not owner then return end
        local ownerName=(typeof(owner)=="Instance" and owner.Name) or (type(owner)=="table" and owner.Name) or "Unknown"
        if not Players:FindFirstChild(ownerName) then return end
        for slot,ad in pairs(al) do
            if type(ad)=="table" then
                local aInfo=AnimalsData[ad.Index]
                if aInfo then
                    local mut=ad.Mutation or "None"
                    if mut=="Yin Yang" then mut="YinYang" end
                    local gv=AnimalsShared:GetGeneration(ad.Index,ad.Mutation,ad.Traits,nil)
                    local gt="$"..NumberUtils:ToString(gv).."/s"
                    table.insert(AAC,{
                        name=aInfo.DisplayName or ad.Index, genText=gt, genValue=gv,
                        mutation=mut, owner=ownerName, plot=plot.Name,
                        slot=tostring(slot), uid=plot.Name.."_"..tostring(slot),
                    })
                end
            end
        end
        table.sort(AAC,function(a,b) return a.genValue>b.genValue end)
    end

    local function setupPlot(plot)
        local ch,tries=nil,0
        while not ch and tries<50 do
            local ok,r=pcall(function() return Synchronizer:Get(plot.Name) end)
            if ok and r then ch=r; break end
            tries+=1; task.wait(0.1)
        end
        if not ch then return end
        scanPlot(plot)
        plot.DescendantAdded:Connect(function() task.wait(0.1); scanPlot(plot) end)
        plot.DescendantRemoving:Connect(function() task.wait(0.1); scanPlot(plot) end)
        task.spawn(function() while plot.Parent do task.wait(5); scanPlot(plot) end end)
    end

    local plots=Workspace:WaitForChild("Plots",8)
    if plots then
        for _,p in ipairs(plots:GetChildren()) do setupPlot(p) end
        plots.ChildAdded:Connect(function(p) task.wait(0.5); setupPlot(p) end)
        plots.ChildRemoved:Connect(function(p)
            lastHash[p.Name]=nil
            for i=#AAC,1,-1 do
                if AAC[i].plot==p.Name then table.remove(AAC,i) end
            end
        end)
    end
end)

                if target then break end
            end
            if not target then
                for _,a in ipairs(AAC) do if a and a.owner~=LocalPlayer.Name then target=a; break end end
            end
        else
            if SPD then target=SPD.animalData end
        end

        if target then
            tNameLbl.Text=string.format("%s  (%s)",target.name,target.genText or "")
            tNameLbl.TextColor3=Theme.Accent; tPStr.Transparency=0.2
        else
